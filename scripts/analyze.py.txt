#!/usr/bin/env python3
"""
=============================================================================
 analyze.py  -  RENEU 2026 PIL-MOF trajectory analysis
 Achinivu Lab, University of Illinois Chicago
=============================================================================

This script does Part C of the research plan (steps 16-18) for you:

  16. radial distribution functions g(r) between the coordinating atoms, and
      first-shell coordination numbers obtained by integrating to the first
      minimum of g(r)
  17. hydrogen-bond population statistics
  18. a summary table you can paste straight into your slides, with each number
      checked against the value the research plan says to expect

You do not edit this file. Run it like this:

    python3 analyze.py --dump dump.lammpstrj --data data.lmp --stage 1

Everything it writes goes into a folder called analysis/ next to the
trajectory: CSV files with the raw numbers, PNG figures, and summary.md.

Only intermolecular pairs are counted. Two atoms in the SAME ion never
contribute to a g(r) or to a hydrogen bond here, because a first solvation
shell is by definition made of neighbours, not of the ion's own atoms.
=============================================================================
"""
import argparse, os, sys, math, csv
import numpy as np

try:
    import matplotlib
    matplotlib.use("Agg")
    import matplotlib.pyplot as plt
    HAVE_PLT = True
except Exception:
    HAVE_PLT = False

# ---------------------------------------------------------------------------
# What the research plan says to expect. analyze.py flags anything outside
# these windows so you notice it before it reaches a slide.
# Section 9 of the research plan; Nd-O window from Li, Song & Merz 2015.
# ---------------------------------------------------------------------------
EXPECTED = {
    "NH_O":  dict(label="N-H...O first peak",      lo=1.6,  hi=2.2,  what="peak"),
    "OH_O":  dict(label="O-H...O first peak",      lo=1.5,  hi=2.2,  what="peak"),
    "N_O":   dict(label="N...O first minimum",     lo=3.0,  hi=4.5,  what="rmin"),
    "M_O":   dict(label="metal-O first peak",      lo=2.2,  hi=2.7,  what="peak"),
    "M_CN":  dict(label="metal first-shell CN",    lo=6.0,  hi=10.0, what="cn"),
}

# Atom-type roles. Types come from the CL&P/RENEU force field.
CAT_N     = {"N1", "N3", "N4"}                 # cation nitrogen
CAT_NH    = {"HN", "H3"}                       # acidic N-H hydrogen
CAT_OH_H  = {"HO"}                             # hydroxyl hydrogen (choline, ethanolammonium)
CAT_OH_O  = {"OH"}
ANION_O   = {"O2"}                             # carboxylate oxygen
ANION_C   = {"CO2", "CO2F", "CO2G", "CO2L"}    # carboxylate carbon
WATER_O   = {"Ow", "OW"}
WATER_H   = {"Hw", "HW"}
METALS    = {"Nd", "La", "Ce", "Pr", "Dy", "Tb", "Sc", "Y", "Li", "Co", "Ni", "Mn"}


# ---------------------------------------------------------------------------
# readers
# ---------------------------------------------------------------------------
def read_type_names(datafile):
    """type id -> force-field type name, from the Masses section of data.lmp."""
    names, inmass = {}, False
    for line in open(datafile):
        s = line.strip()
        if s.startswith("Masses"):
            inmass = True
            continue
        if inmass:
            if not s:
                if names:
                    break
                continue
            if "#" not in s:
                break
            body, comment = s.split("#", 1)
            tok = body.split()
            if not tok or not tok[0].isdigit():
                break
            names[int(tok[0])] = comment.strip().split()[0]
    if not names:
        sys.exit("ERROR: could not read atom type names from " + datafile)
    return names


def frames(dumpfile):
    """Yield (step, box_lengths, box_lo, arrays) for each frame of a LAMMPS dump."""
    f = open(dumpfile)
    while True:
        line = f.readline()
        if not line:
            return
        if not line.startswith("ITEM: TIMESTEP"):
            continue
        step = int(f.readline())
        f.readline()                                  # ITEM: NUMBER OF ATOMS
        natom = int(f.readline())
        f.readline()                                  # ITEM: BOX BOUNDS
        lo, L = [], []
        for _ in range(3):
            a, b = [float(x) for x in f.readline().split()[:2]]
            lo.append(a)
            L.append(b - a)
        hdr = f.readline().split()[2:]                # ITEM: ATOMS id mol type ...
        first = f.readline().split()

        def isnum(v):
            try:
                float(v)
                return True
            except ValueError:
                return False

        num = [i for i, v in enumerate(first) if isnum(v)]   # skip 'element' etc.
        cols = {hdr[i]: k for k, i in enumerate(num)}
        raw = np.empty((natom, len(num)))
        raw[0] = [float(first[i]) for i in num]
        for i in range(1, natom):
            t = f.readline().split()
            raw[i] = [float(t[j]) for j in num]
        order = np.argsort(raw[:, cols["id"]])
        raw = raw[order]
        yield step, np.array(L), np.array(lo), raw, cols


def wrap(x, lo, L):
    return (x - lo) % L


# ---------------------------------------------------------------------------
# geometry
# ---------------------------------------------------------------------------
def mindist(a, b, L):
    """Minimum-image distance matrix between coordinate sets a (n,3) and b (m,3)."""
    d = a[:, None, :] - b[None, :, :]
    d -= L * np.round(d / L)
    return np.sqrt((d * d).sum(-1))


def rdf_frame(xa, xb, mola, molb, L, rmax, nbins, same_set):
    """Histogram of intermolecular a-b distances for one frame."""
    if len(xa) == 0 or len(xb) == 0:
        return np.zeros(nbins), 0
    d = mindist(xa, xb, L)
    mask = mola[:, None] != molb[None, :]          # drop same-molecule pairs
    if same_set:
        np.fill_diagonal(mask, False)
    dd = d[mask]
    h, _ = np.histogram(dd, bins=nbins, range=(0.0, rmax))
    return h, mask.sum()


def compute_rdf(traj, ia, ib, rmax=12.0, nbins=240, same_set=False):
    """Average g(r) and running coordination number n(r) over all frames."""
    hist = np.zeros(nbins)
    nA = nB = nfr = 0
    volsum = 0.0
    for L, xyz, mol in traj:
        rmaxok = min(rmax, L.min() / 2.0)
        h, _ = rdf_frame(xyz[ia], xyz[ib], mol[ia], mol[ib], L, rmax, nbins, same_set)
        hist += h
        nA, nB = len(ia), len(ib)
        volsum += float(np.prod(L))
        nfr += 1
    if nfr == 0 or nA == 0 or nB == 0:
        return None
    edges = np.linspace(0.0, rmax, nbins + 1)
    r = 0.5 * (edges[1:] + edges[:-1])
    shell = 4.0 / 3.0 * math.pi * (edges[1:] ** 3 - edges[:-1] ** 3)
    vol = volsum / nfr
    rho_b = nB / vol
    ideal = shell * rho_b * nA * nfr
    g = np.divide(hist, ideal, out=np.zeros_like(hist), where=ideal > 0)
    # running coordination number: average number of B within r of one A
    n_r = np.cumsum(hist) / (nA * nfr)
    if same_set:                                   # do not double count A-A pairs
        pass
    return dict(r=r, g=g, n=n_r, rmax_used=min(rmax, rmaxok), nA=nA, nB=nB, nframes=nfr)


def first_peak_min(r, g, rlow=1.2):
    """First maximum of g(r) above rlow, and the first minimum after it."""
    k = np.ones(5) / 5.0
    gs = np.convolve(g, k, mode="same")
    i0 = np.searchsorted(r, rlow)
    if i0 >= len(r) - 3:
        return None, None, None
    seg = gs[i0:]
    ipk = int(np.argmax(seg)) + i0
    if gs[ipk] < 1.0e-6:
        return None, None, None
    j = ipk
    while j < len(gs) - 1 and gs[j + 1] <= gs[j]:
        j += 1
    if j >= len(gs) - 2:
        return r[ipk], gs[ipk], None
    return r[ipk], gs[ipk], r[j]


# ---------------------------------------------------------------------------
# hydrogen bonds
# ---------------------------------------------------------------------------
def find_DH(types, mol, xyz0, L, donors, hydrogens):
    """Covalently bonded donor-hydrogen pairs, found once from the first frame."""
    idD = np.where(np.isin(types, list(donors)))[0]
    idH = np.where(np.isin(types, list(hydrogens)))[0]
    pairs = []
    if len(idD) == 0 or len(idH) == 0:
        return pairs
    d = mindist(xyz0[idD], xyz0[idH], L)
    for i, di in enumerate(idD):
        for j, hj in enumerate(idH):
            if d[i, j] < 1.35 and mol[di] == mol[hj]:
                pairs.append((di, hj))
    return pairs


def hbond_stats(traj, dh_pairs, acceptors, rHA=2.5, angmin=130.0):
    """Mean number of hydrogen bonds per frame and per donor."""
    if not dh_pairs or len(acceptors) == 0:
        return None
    D = np.array([p[0] for p in dh_pairs])
    H = np.array([p[1] for p in dh_pairs])
    counts = []
    for L, xyz, mol in traj:
        xd, xh, xa = xyz[D], xyz[H], xyz[acceptors]
        d = mindist(xh, xa, L)
        ok = d < rHA
        ok &= mol[H][:, None] != mol[acceptors][None, :]
        if not ok.any():
            counts.append(0.0)
            continue
        ii, jj = np.where(ok)
        v1 = xd[ii] - xh[ii]
        v2 = xa[jj] - xh[ii]
        v1 -= L * np.round(v1 / L)
        v2 -= L * np.round(v2 / L)
        cs = (v1 * v2).sum(1) / (np.linalg.norm(v1, axis=1) * np.linalg.norm(v2, axis=1))
        ang = np.degrees(np.arccos(np.clip(cs, -1, 1)))
        counts.append(float((ang > angmin).sum()))
    counts = np.array(counts)
    return dict(mean=counts.mean(), std=counts.std(), per_donor=counts.mean() / len(D),
                ndonor=len(D), rHA=rHA, angmin=angmin)


# ---------------------------------------------------------------------------
# main
# ---------------------------------------------------------------------------
def main():
    ap = argparse.ArgumentParser(description="RENEU 2026 trajectory analysis")
    ap.add_argument("--dump", default="dump.lammpstrj")
    ap.add_argument("--data", default="data.lmp")
    ap.add_argument("--stage", type=int, default=1, choices=[1, 2, 3])
    ap.add_argument("--out", default="analysis")
    ap.add_argument("--skip", type=float, default=0.2,
                    help="fraction of frames to discard at the start (default 0.2)")
    ap.add_argument("--rmax", type=float, default=12.0)
    ap.add_argument("--title", default="")
    a = ap.parse_args()

    for f in (a.dump, a.data):
        if not os.path.exists(f):
            sys.exit(f"ERROR: {f} not found. Did the simulation finish? "
                     "Run the stage script again and read the last lines of log.lammps.")
    os.makedirs(a.out, exist_ok=True)

    names = read_type_names(a.data)
    print(f"  force-field types in this system: {' '.join(sorted(set(names.values())))}")

    raw_frames = list(frames(a.dump))
    if not raw_frames:
        sys.exit("ERROR: no frames in " + a.dump)
    nskip = int(len(raw_frames) * a.skip)
    used = raw_frames[nskip:]
    print(f"  {len(raw_frames)} frames read, first {nskip} discarded as equilibration, "
          f"{len(used)} analysed")

    step0, L0, lo0, raw0, cols = used[0]
    tid = raw0[:, cols["type"]].astype(int)
    types = np.array([names[t] for t in tid])
    mol = raw0[:, cols["mol"]].astype(int)

    traj = []
    for step, L, lo, raw, cols in used:
        xyz = np.column_stack([wrap(raw[:, cols["xu"]], lo[0], L[0]),
                               wrap(raw[:, cols["yu"]], lo[1], L[1]),
                               wrap(raw[:, cols["zu"]], lo[2], L[2])])
        traj.append((L, xyz, mol))

    sel = lambda s: np.where(np.isin(types, list(s)))[0]
    idx = dict(catN=sel(CAT_N), catNH=sel(CAT_NH), catOHH=sel(CAT_OH_H),
               catOHO=sel(CAT_OH_O), anO=sel(ANION_O), anC=sel(ANION_C),
               wO=sel(WATER_O), wH=sel(WATER_H), metal=sel(METALS))

    # ---- which g(r) to compute, per stage ---------------------------------
    pairs = []
    if len(idx["catNH"]):
        pairs.append(("N-H(cation)", idx["catNH"], "O(carboxylate)", idx["anO"], "NH_O"))
    if len(idx["catOHH"]):
        pairs.append(("O-H(cation)", idx["catOHH"], "O(carboxylate)", idx["anO"], "OH_O"))
    # the N...O window in the plan assumes an acidic N-H. A quaternary cation
    # (choline) has no N-H, so its nitrogen sits further from the anion by
    # construction and the window does not apply.
    pairs.append(("N(cation)", idx["catN"], "O(carboxylate)", idx["anO"],
                  "N_O" if len(idx["catNH"]) else None))
    pairs.append(("N(cation)", idx["catN"], "C(carboxylate)", idx["anC"], None))
    if a.stage >= 2 and len(idx["wO"]):
        pairs.append(("O(water)", idx["wO"], "O(carboxylate)", idx["anO"], None))
        pairs.append(("O(water)", idx["wO"], "N(cation)", idx["catN"], None))
        pairs.append(("O(water)", idx["wO"], "O(water)", idx["wO"], None))
    if a.stage >= 3 and len(idx["metal"]):
        # the metal-water pair is the parameter validation (Stage 3a);
        # the metal-carboxylate pair is the science question (Stage 3b), and it
        # has no plan window because that is what you are trying to measure.
        pairs.append(("metal ion", idx["metal"], "O(carboxylate)", idx["anO"], None))
        if len(idx["wO"]):
            pairs.append(("metal ion", idx["metal"], "O(water)", idx["wO"], "M_O"))

    rows, flags = [], []
    for lblA, ia, lblB, ib, key in pairs:
        if len(ia) == 0 or len(ib) == 0:
            continue
        same = (lblA == lblB)
        res = compute_rdf(traj, ia, ib, rmax=a.rmax, same_set=same)
        if res is None:
            continue
        rpk, gpk, rmin = first_peak_min(res["r"], res["g"])
        cn = float(np.interp(rmin, res["r"], res["n"])) if rmin else float("nan")
        tag = f"{lblA}_{lblB}".replace("(", "").replace(")", "").replace(" ", "").replace("-", "")
        with open(os.path.join(a.out, f"rdf_{tag}.csv"), "w", newline="") as fh:
            w = csv.writer(fh)
            w.writerow(["r_angstrom", "g_r", "running_coordination_number"])
            for rr, gg, nn in zip(res["r"], res["g"], res["n"]):
                w.writerow([f"{rr:.4f}", f"{gg:.5f}", f"{nn:.5f}"])
        rows.append(dict(pair=f"{lblA} ... {lblB}", nA=res["nA"], nB=res["nB"],
                         rpeak=rpk, gpeak=gpk, rmin=rmin, cn=cn, key=key))
        if HAVE_PLT:
            fig, ax1 = plt.subplots(figsize=(5.2, 3.6), dpi=150)
            ax1.plot(res["r"], res["g"], lw=1.6, color="#1f3b73")
            ax1.set_xlabel(r"$r$ / $\AA$")
            ax1.set_ylabel(r"$g(r)$", color="#1f3b73")
            ax1.axhline(1.0, color="grey", lw=0.6, ls=":")
            if rpk:
                ax1.axvline(rpk, color="#c0392b", lw=0.8, ls="--")
                ax1.annotate(f"peak {rpk:.2f} " + r"$\AA$", (rpk, gpk),
                             textcoords="offset points", xytext=(6, -2), fontsize=7,
                             color="#c0392b")
            if rmin:
                ax1.axvline(rmin, color="#117a3d", lw=0.8, ls="--")
                ax1.annotate(f"first min {rmin:.2f} " + r"$\AA$", (rmin, 1.0),
                             textcoords="offset points", xytext=(6, 18), fontsize=7,
                             color="#117a3d")
            ax2 = ax1.twinx()
            ax2.plot(res["r"], res["n"], lw=1.2, color="#e08a00")
            ax2.set_ylabel("running CN", color="#e08a00")
            ax1.set_title(f"{a.title or 'RDF'}: {lblA} - {lblB}", fontsize=9)
            fig.tight_layout()
            fig.savefig(os.path.join(a.out, f"fig_rdf_{tag}.png"))
            plt.close(fig)

    # ---- hydrogen bonds ---------------------------------------------------
    donors = CAT_N | CAT_OH_O | WATER_O
    hyd = CAT_NH | CAT_OH_H | WATER_H
    dh = find_DH(types, mol, traj[0][1], traj[0][0], donors, hyd)
    acc = sel(ANION_O | CAT_OH_O | WATER_O)
    hb = hbond_stats(traj, dh, acc)

    # ---- density ----------------------------------------------------------
    dens = None
    dfile = os.path.join(os.path.dirname(os.path.abspath(a.dump)), "density.dat")
    if os.path.exists(dfile):
        vals = [float(l.split()[1]) for l in open(dfile)
                if l.strip() and not l.startswith("#") and len(l.split()) >= 2]
        if vals:
            keep = vals[len(vals) // 5:]
            dens = (float(np.mean(keep)), float(np.std(keep)))

    # ---- summary ----------------------------------------------------------
    with open(os.path.join(a.out, "coordination_numbers.csv"), "w", newline="") as fh:
        w = csv.writer(fh)
        w.writerow(["pair", "n_atoms_A", "n_atoms_B", "first_peak_r_A",
                    "g_at_peak", "first_minimum_r_A", "first_shell_CN"])
        for r in rows:
            w.writerow([r["pair"], r["nA"], r["nB"],
                        f"{r['rpeak']:.3f}" if r["rpeak"] else "",
                        f"{r['gpeak']:.3f}" if r["rpeak"] else "",
                        f"{r['rmin']:.3f}" if r["rmin"] else "",
                        f"{r['cn']:.3f}" if r["cn"] == r["cn"] else ""])

    md = []
    md.append(f"# Analysis summary - {a.title or os.path.basename(os.getcwd())}")
    md.append("")
    md.append(f"Stage {a.stage}. {len(used)} production frames analysed "
              f"(first {nskip} of {len(raw_frames)} discarded).")
    md.append("")
    if dens:
        md.append(f"**Table 1. Equilibrated density from the NPT production run.**")
        md.append("")
        md.append("| quantity | value |")
        md.append("|---|---|")
        md.append(f"| mean density | {dens[0]:.4f} g/cm3 |")
        md.append(f"| standard deviation | {dens[1]:.4f} g/cm3 |")
        md.append("")
        if dens[0] < 0.5:
            md.append("> CHECK: a density below 0.5 g/cm3 means the box has not "
                      "condensed. Do not analyse this trajectory. See TROUBLESHOOTING.md.")
            md.append("")
    if len(idx["catN"]) and len(idx["catNH"]) == 0:
        md.append("> Note: this cation has no acidic N-H. The N...O window in "
                  "Section 9 of the research plan assumes one, so no plan check is "
                  "applied to the N...O row; the hydroxyl O-H is the donor here.")
        md.append("")
    md.append("**Table 2. First-shell structure from the radial distribution functions.**")
    md.append("")
    md.append("| atom pair | first peak / A | g at peak | first minimum / A | "
              "first-shell CN | plan check |")
    md.append("|---|---|---|---|---|---|")
    for r in rows:
        chk = ""
        if r["key"] in EXPECTED and r["rpeak"]:
            e = EXPECTED[r["key"]]
            val = r["rpeak"] if e["what"] == "peak" else (
                  r["rmin"] if e["what"] == "rmin" else r["cn"])
            if val is None or val != val:
                chk = "not determined"
            elif e["lo"] <= val <= e["hi"]:
                chk = f"as expected ({e['label']} {e['lo']}-{e['hi']})"
            else:
                chk = f"OUTSIDE expected {e['lo']}-{e['hi']} - explain or re-check"
        md.append(f"| {r['pair']} | "
                  f"{r['rpeak']:.2f} | {r['gpeak']:.2f} | "
                  f"{r['rmin']:.2f} | {r['cn']:.2f} | {chk} |"
                  if r["rpeak"] and r["rmin"] else
                  f"| {r['pair']} | - | - | - | - | first shell not resolved |")
    md.append("")
    if hb:
        md.append("**Table 3. Hydrogen-bond population "
                  f"(H...acceptor < {hb['rHA']} A, D-H...A angle > {hb['angmin']} deg).**")
        md.append("")
        md.append("| quantity | value |")
        md.append("|---|---|")
        md.append(f"| donor D-H groups in the box | {hb['ndonor']} |")
        md.append(f"| mean hydrogen bonds per frame | {hb['mean']:.1f} +/- {hb['std']:.1f} |")
        md.append(f"| mean hydrogen bonds per donor | {hb['per_donor']:.3f} |")
        md.append("")
    if a.stage >= 3 and len(idx["metal"]):
        md.append("> Stage 3 reading order: first confirm the metal-O(water) row "
                  "(expect a peak near 2.4-2.5 A and a coordination number near 8-9). "
                  "Only if that holds does the metal-O(carboxylate) row mean anything. "
                  "A carboxylate that has not entered the first shell shows up as a "
                  "peak beyond about 5 A with a coordination number near zero, which "
                  "usually means the run is too short rather than that the anion does "
                  "not coordinate.")
        md.append("")
    md.append("**Figure titles.** One PNG per pair is written next to this file:")
    for r in rows:
        md.append(f"- Radial distribution function and running coordination number, "
                  f"{r['pair']}.")
    md.append("")
    md.append("Raw numbers behind every figure are in the rdf_*.csv files, so any "
              "value you quote can be traced to a file (research plan Section 5.2).")
    open(os.path.join(a.out, "summary.md"), "w").write("\n".join(md) + "\n")

    print("\n" + "\n".join(md[:40]))
    print(f"\n  written to {os.path.abspath(a.out)}/")


if __name__ == "__main__":
    main()
